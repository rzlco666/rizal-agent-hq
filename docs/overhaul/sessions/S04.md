```rust
#[test]
fn parses_ps_and_redacts_args() {
    let p = parse_ps(include_str!("fixtures/ps.txt"));
    assert_eq!(p.len(), 5);
    assert_eq!(p[0].ppid, 1);
    assert!(!p.iter().any(|x| x.args.contains("abc.def.ghi")));

#[test]

    let out = run_timeout("sleep", &["5"], 100);
    assert!(out.is_none());
```

### Task 3 — Presence agent L1

- [ ] **Step 1: Tulis tes gagal** di `tests/presence.rs`

```rust
#[tokio::test]
async fn omp_process_creates_l1_agent() {
    let ctx = ctx_with_procs(vec![proc("omp", "omp --model x")]);
    let out = ProcessPresenceAdapter::default().observe(&ctx).await.unwrap();
    let a = out.iter().find(|d| d.subject.key() == "agent:omp").unwrap();
    assert_eq!(a.kind, EventKind::AgentObserved);
    assert_eq!(a.payload["status"], "bekerja");
    assert_eq!(a.payload["evidence"], "L1");
}

#[tokio::test]
async fn l1_agent_never_claims_a_task() {
    let ctx = ctx_with_procs(vec![proc("cursor", "cursor")]);
    let out = ProcessPresenceAdapter::default().observe(&ctx).await.unwrap();
    assert!(!serde_json::to_string(&out).unwrap().contains("task."));
}

#[tokio::test]
async fn collector_survives_disappearing_source() {
    let ctx = ctx_with_procs(vec![proc("omp", "omp")]);
    let adapter = ProcessPresenceAdapter { seen: Mutex::new(HashSet::new()) };
    adapter.observe(&ctx).await.unwrap();
    let ctx2 = ctx_with_procs(vec![]);
    let out = adapter.observe(&ctx2).await.unwrap();
    assert!(out.iter().any(|d| d.payload["status"] == "offline"));
}
```

- [ ] **Step 2: Jalankan, pastikan gagal.**

- [ ] **Step 3: Implementasi** `runtime/presence.rs` + `runtime/mod.rs`

`RuntimeAdapter` sesuai `01` dan `02`. `classify` memetakan proses ke keluarga:
omp, cursor, claude, codex, devin, gemini, opencode, copilot, mcp. Satu keluarga
= satu `agent.observed` L1: status `bekerja` bila proses ada, `offline` setelah
2 siklus tidak terlihat. Activity memakai `Sentence` key `agent.process_running`
dengan param `family`; tidak ada klaim task. `runtime_key` =
`presence:<family>` sampai adapter dalam mengambil alih.

- [ ] **Step 4: Jalankan, pastikan lulus.**

- [ ] **Step 5: Commit** — `feat(collect): L1 process presence agents`
