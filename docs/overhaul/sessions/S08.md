test("applies patch and tracks seq", async () => {
  const s = createLiveStore(fakeSocket());

test("snapshot_required triggers refetch", async () => {

test("reconnect uses last seq and backs off", async () => {
  const s = createLiveStore(fakeSocket());
  s.apply({ type: "hello", seq: 42 });
  s.disconnect();
  expect(s.nextUrl()).toContain("since=42");
  expect(s.backoffMs()).toBeGreaterThanOrEqual(250);
});
