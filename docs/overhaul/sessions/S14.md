### Task 1 — Nav grid dan A*

- [ ] **Step 1: Tulis tes gagal** `nav.test.ts`

```ts
test("shortest path on open grid", () => {
  const g = NavGrid.open(20, 20, 0.5);
  const p = findPath(g, { x: 1, z: 1 }, { x: 9, z: 1 });
  expect(p.length).toBe(9);
  expect(p.at(-1)).toEqual({ x: 9, z: 1 });
});

test("avoids blocked cells", () => {
  const g = NavGrid.open(10, 10, 0.5).block(5, 0, 5, 9);
  const p = findPath(g, { x: 2, z: 2 }, { x: 8, z: 2 });
  expect(p.some(c => c.x === 5)).toBe(false);
  expect(p.length).toBeGreaterThan(7);
});

test("smoothing keeps endpoints and removes collinear points", () => {
  const p = smooth([{ x: 0, z: 0 }, { x: 1, z: 0 }, { x: 2, z: 0 }], g());
  expect(p).toEqual([{ x: 0, z: 0 }, { x: 2, z: 0 }]);
});
```

### Task 4 — Spawn, walk, arrival

- [ ] **Step 1: Tulis tes gagal** `agents.test.ts`

```ts
test("new agent spawns at entrance and walks to desk", () => {
  const sim = AgentSim.create(layoutAgents([agent("agent:omp")]), 0);
  expect(sim.agents[0].position).toEqual(sim.entrance);
  for (let i = 0; i < 600; i++) sim.step(1 / 60);
  expect(sim.agents[0].position).toEqual(sim.deskPosition("agent:omp"));
  expect(sim.agents[0].anim).toBe("type");
});

test("removed agent walks out and disappears", () => {
  const sim = AgentSim.create(layoutAgents([agent("agent:omp")]), 0);
  sim.stepUntilArrived();
  sim.sync([]);
  expect(sim.agents[0].leaving).toBe(true);
  sim.stepUntilGone();
  expect(sim.agents).toHaveLength(0);
});
```
