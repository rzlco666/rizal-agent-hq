```rust
#[tokio::test]
async fn declarative_poll_maps_items_to_events() {
    let srv = mock_poll(json!({"channels":[{"id":"C1","name":"general"}]})).await;
    let c = DeclarativeConnector::from_yaml_file("tests/fixtures/slack.yaml", secrets()).await.unwrap();
    let out = c.poll(&ctx(&srv.uri())).await.unwrap();
    assert_eq!(out.len(), 1);
    assert_eq!(out[0].kind, EventKind::ConnectorEvent);
    assert_eq!(out[0].payload["event_type"], "channel");
}

#[tokio::test]
async fn webhook_rejects_bad_signature_and_replay() {
    let c = webhook_connector("hmac-secret");
    assert!(c.handle_webhook(req_with_sig("bad", now())).await.is_err());
    assert!(c.handle_webhook(req_with_sig(valid(), now() - 10 * 60_000)).await.is_err());
    assert!(c.handle_webhook(req_with_sig(valid(), now())).await.is_ok());
    assert!(c.handle_webhook(req_with_sig(valid(), now())).await.is_err()); // replay
}
```

- [ ] **Step 2: Jalankan, pastikan gagal.**

- [ ] **Step 3: Implementasi** `declarative/*`

Auth: none/bearer/basic/api-key/OAuth2 client-credentials + authorization-code
(callback disimpan di state). Poll: URL, query, pagination (cursor/page/next),
JSONPath items, mapping ke `ConnectorEvent` (`subject_hint`, `event_type`,
`data`). Webhook: HMAC SHA-256 header/timestamp/nonce, window 5 menit,
idempotency key dari signature+delivery id. Actions: method/url/headers/body
template, `${input.x}` dan `${secret:x}`, timeout 15 s, response dibatasi 256 KB.
Semua error punya kode (`auth`, `schema`, `rate_limit`, `http`, `timeout`).
