#[tokio::test]
async fn ingest_redacts_before_persist() {
    let (store, _dir) = test_store().await;
    let ing = Ingestor::new(store.clone());
    let mut d = log_with_raw("agent:omp", "menjalankan tes", "token ghp_abcdefghijklmnop");
    d.payload["sentence"]["params"]["detail"] = "Authorization: Bearer abc.def.ghi".into();
    let r = ing.ingest(vec![d]).await.unwrap();
    let ev = store.read_since(Seq(0), 10).await.unwrap();
    let text = serde_json::to_string(&ev[0].draft.payload).unwrap();
    assert!(!text.contains("ghp_"));
    assert!(!text.contains("abc.def.ghi"));
    assert_eq!(r.accepted.len(), 1);
}

#[tokio::test]
async fn duplicate_webhook_is_noop() {

    let d = webhook_draft("github", "delivery-1");
    let a = ing.ingest(vec![d.clone()]).await.unwrap();
    let b = ing.ingest(vec![d]).await.unwrap();
    assert_eq!(a.accepted.len(), 1);
    assert_eq!(b.accepted.len(), 0);
    assert_eq!(b.duplicates, 1);

- [ ] **Step 3: Implementasi** `ingest.rs`

```rust
pub struct Ingestor { store: Store }
pub struct IngestReport { pub accepted: Vec<Event>, pub duplicates: usize }

impl Ingestor {
    pub fn new(store: Store) -> Self;
    pub async fn ingest(&self, mut drafts: Vec<EventDraft>) -> Result<IngestReport, StoreError>;
}
```

Urutan: `validate` → `redact_json(payload)` → `store.append`. Hitung duplikat
dari selisih jumlah draft vs event yang diterima. Jangan menelan error validasi:
satu draft invalid membatalkan batch (atomic), dan error menyebut subject+kind.

### Task 3 — WebSocket + SSE

- [ ] **Step 1: Tulis tes gagal** di `crates/hq-api/tests/ws.rs`

```rust
#[tokio::test]
async fn ws_sends_hello_then_patch() {
    let (addr, ing, _dir) = spawn_test_api().await;
    let (mut ws, _) = tokio_tungstenite::connect_async(format!("ws://{addr}/ws?since=0")).await.unwrap();
    let hello = next_json(&mut ws).await;
    assert_eq!(hello["type"], "hello");
    ing.ingest(vec![agent_observed("agent:omp", "bekerja")]).await.unwrap();
    let msg = next_json(&mut ws).await;
    assert_eq!(msg["type"], "patch");
    assert!(msg["seq"].as_i64().unwrap() > 0);
}

#[tokio::test]
async fn ws_reconnect_replays_missed_patches() {
    let (addr, ing, _dir) = spawn_test_api().await;
    ing.ingest(vec![heartbeat("agent:muse", "a")]).await.unwrap();
    let seq = ing.store().last_seq().await.unwrap();
    ing.ingest(vec![heartbeat("agent:muse", "b")]).await.unwrap();
    let (mut ws, _) = tokio_tungstenite::connect_async(
        format!("ws://{addr}/ws?since={}", seq.0)).await.unwrap();
    assert_eq!(next_json(&mut ws).await["type"], "hello");
    assert_eq!(next_json(&mut ws).await["type"], "patch");
}

#[tokio::test]
async fn stale_since_asks_for_snapshot() {
    let (addr, _ing, _dir) = spawn_test_api().await;
    let (mut ws, _) = tokio_tungstenite::connect_async(
        format!("ws://{addr}/ws?since=1")).await.unwrap();
    assert_eq!(next_json(&mut ws).await["type"], "snapshot_required");
}
```

- [ ] **Step 2: Jalankan, pastikan gagal.**

- [ ] **Step 3: Implementasi** `ws.rs` + `sse.rs` + `router.rs`

`api_router` memasang `/health`, `/ws`, `/api/v1/events/stream`. WS hanya
menerima peer loopback pada sesi ini; non-loopback → 401 (ticket ditambahkan
S06). Pesan:

```json
{"type":"hello","seq":42}
{"type":"patch","seq":43,"upserts":[...],"deletes":[]}
{"type":"snapshot_required","seq":43}
```

Kirim `hello` dengan seq terakhir, lalu patch dari ring; setelah itu streaming
dari broadcast. SSE menulis `id: <seq>\ndata: <patch-json>\n\n`; heartbeat
komentar setiap 15 detik.

- [ ] **Step 4: Jalankan, pastikan lulus.**
