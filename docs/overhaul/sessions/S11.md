#[tokio::test]
async fn snooze_expires_and_reopens() {
    let clock = TestClock::at(1_000_000);
    let (store, _dir) = test_store().await;
    let eng = test_engine(store.clone(), clock.clone());
    eng.snooze("a1", 60, "owner").await.unwrap();
    clock.advance(61_000);
    let drafts = eng.tick().await.unwrap();
    assert!(drafts.iter().any(|d| d.kind == EventKind::AlertUpdated));
    assert_eq!(store.alert("a1").await.unwrap().unwrap().state, AlertState::Open);
}

#[tokio::test]
async fn recovery_resolves_after_two_clean_cycles() {
    let (store, _dir) = test_store().await;
    let eng = test_engine(store.clone(), TestClock::at(1_000_000));
    seed_open_alert(&store).await;
    assert!(eng.tick().await.unwrap().is_empty()); // clean 1
    let d = eng.tick().await.unwrap();             // clean 2
    assert!(d.iter().any(|x| x.kind == EventKind::AlertResolved));
}
