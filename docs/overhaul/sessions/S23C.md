### Task 4 — Markets: primary dan fallback

- [ ] **Step 1: Tulis tes gagal**

```rust
#[tokio::test]
async fn eodhd_primary_and_fmp_fallback() {
    let eodhd = mock_eodhd().ok("NVDA.US", 186.64).await;
    let fmp = mock_fmp().ok("SAP.DE", 210.10).await;
    let c = MarketsConnector::test(eodhd.uri(), Some(fmp.uri()));
    let out = c.poll(&ctx()).await.unwrap();
    let nvda = out.iter().find(|e| e.payload["symbol"] == "NVDA.US").unwrap();
    assert_eq!(nvda.payload["source"], "eodhd");
    let sap = out.iter().find(|e| e.payload["symbol"] == "SAP.DE").unwrap();
    assert_eq!(sap.payload["source"], "fmp");
}

#[tokio::test]
async fn unauthorized_and_weekend_are_honest() {
    let c = MarketsConnector::test(mock_401().uri(), None);
    assert_eq!(c.poll(&ctx()).await.unwrap_err().code(), "auth");
    let weekend = MarketsConnector::test(mock_empty().uri(), None);
    assert!(weekend.poll(&ctx_weekend()).await.unwrap().is_empty());
}
```
