### Task 6 — Halaman `/settings` dan endpoint device/backup

- [ ] **Step 1: Tulis tes gagal**

```rust
#[tokio::test]
async fn settings_roundtrip_and_device_revoke() {
    let app = app_with_device();
    let j: serde_json::Value = body_json(get(app.clone(), "/api/v1/settings", loopback()).await).await;
    assert_eq!(j["locale"], "id");
    let p = patch(app.clone(), "/api/v1/settings", json!({"locale":"en","retention_days":90}), loopback()).await;
    assert_eq!(p.status(), 200);
    assert_eq!(delete(app.clone(), "/api/v1/devices/dev1", loopback()).await.status(), 204);
    assert_eq!(get_with_cookie(app, "/api/v1/agents").await.status(), 401);
}

#[tokio::test]
async fn backup_endpoint_reports_age_and_path() {
    let app = app_with_backup();
    let j: serde_json::Value = body_json(get(app, "/api/v1/backup", loopback()).await).await;
    assert!(j["last_backup_ms"].as_i64().is_some());
    assert!(j["path"].as_str().unwrap().ends_with(".db"));
}
```
